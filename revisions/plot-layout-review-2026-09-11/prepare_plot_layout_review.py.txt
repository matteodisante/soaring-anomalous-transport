def sha(data): return hashlib.sha256(data).hexdigest()
def module(name,source):
 ns={'__name__':'layout_review_module','__file__':str(ROOT/name)}
 exec(compile(source,str(ROOT/name),'exec'),ns)
 return ns

def capture(fig):
 """Data-space values and styles, independent of panel geometry and text position."""
 result=[]
 for ax in fig.axes:
  result.append({
   'title':ax.get_title(),'ylabel':ax.get_ylabel(),'xscale':ax.get_xscale(),'yscale':ax.get_yscale(),
   'xlim':list(ax.get_xlim()),'ylim':list(ax.get_ylim()),
   'lines':[{'x':np.asarray(l.get_xdata()).tolist(),'y':np.asarray(l.get_ydata()).tolist(),
    'color':l.get_color(),'label':l.get_label(),'style':l.get_linestyle(),'width':l.get_linewidth(),
    'marker':l.get_marker()} for l in ax.lines],
   'collections':[{'paths':[p.vertices.tolist() for p in c.get_paths()],
     'offsets':c.get_offsets().tolist(),'facecolor':c.get_facecolor().tolist(),
     'edgecolor':c.get_edgecolor().tolist()} for c in ax.collections],
   'patches':[{'type':type(p).__name__,'path':p.get_path().vertices.tolist(),
     'center':list(p.center) if hasattr(p,'center') else None,
     'width':getattr(p,'width',None),'height':getattr(p,'height',None),
     'angle':getattr(p,'angle',None),'label':p.get_label()} for p in ax.patches],
   'annotations':[t.get_text() for t in ax.texts],
  })
 return result

for fn in ['draw_by_discipline','draw_by_terrain','draw_by_level']:
 tree=ast.parse(kin_after); node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name==fn)
 lines=kin_after.splitlines(keepends=True); index=node.end_lineno-1

 lines[index:index]=['    for ax in axes[:-1]:\n','        ax.set_xlabel("")\n']
 kin_after=''.join(lines)
rev_before=(ROOT/REV).read_text()
rev_after=rev_before.replace('''    """Render at the thesis text width: every label is at least nine points."""''',
 '''    """Render at thesis text width with the shared scientific figure style."""''')
old='''        fig.savefig(OUT / filename, metadata=PDF_META)'''
new='''        # Include every label in the PDF even when backend text extents differ.

                'unchanged_numerical_ast_sha256':sha(ast.dump(before_tree,include_attributes=False).encode())}
