"""Complete the saved render comparison, allowing only roundoff in automatic axis limits."""
from pathlib import Path
import ast,json,hashlib
import numpy as np
D=Path('/tmp/plot-layout-review');R=Path('/Users/matteodisante/local_master_thesis/soaring-anomalous-transport')
a=json.loads((D/'capture-before.json').read_text());b=json.loads((D/'capture-after.json').read_text())
axis_changes=[]
for filename in a:
 for i,(ax,bx) in enumerate(zip(a[filename],b[filename],strict=True)):
  for key in ['xlim','ylim']:
   assert np.allclose(ax[key],bx[key],rtol=0,atol=1e-12),(filename,key)
   if ax[key]!=bx[key]:axis_changes.append({'figure':filename,'axis':i,'key':key,'before':ax[key],'after':bx[key]})
   ax.pop(key);bx.pop(key)
assert json.dumps(a,sort_keys=True)==json.dumps(b,sort_keys=True),'A curve, band, covariance ellipse or label changed'
sha=lambda data:hashlib.sha256(data).hexdigest()
files={}
for path in (D/'before/scripts').rglob('*.py'):
 name=path.relative_to(D/'before').as_posix();before=path.read_bytes();after=(D/'after'/name).read_bytes()
 assert (R/name).read_bytes()==before
 allowed=['draw_by_discipline','draw_by_terrain','draw_by_level'] if 'kinematic' in name else ['draw']
 trees=[ast.parse(x) for x in [before,after]]
 for t in trees:
  for n in t.body:
   if isinstance(n,ast.FunctionDef) and n.name in allowed:n.body=[ast.Pass()]
 assert ast.dump(trees[0],include_attributes=False)==ast.dump(trees[1],include_attributes=False)
 files[name]={'before_sha256':sha(before),'after_sha256':sha(after),'changed_drawing_functions':allowed,
  'unchanged_numerical_ast_sha256':sha(ast.dump(trees[0],include_attributes=False).encode())}
outputs={}
for name in a:
 before=(D/'before/rendered'/name).read_bytes();after=(D/'after/rendered'/name).read_bytes()
 assert before==(R/'thesis/generated'/name).read_bytes(),('baseline differs',name)
 outputs[name]={'before_sha256':sha(before),'after_sha256':sha(after)}
manifest={'run_id':'20260910T221820Z-05d36ab4','files':files,'outputs':outputs,
 'unchanged_plot_data_sha256':sha(json.dumps(a,sort_keys=True).encode()),
 'automatic_axis_limit_roundoff':axis_changes,
 'scope':'Exact identity of data-space curves, bands, ellipse parameters, styles and labels; automatic axis limits agree within 1e-12 (one 2e-16 difference). Baseline PDFs reproduce current execution byte for byte. Numerical source AST is unchanged outside the listed drawing functions. Changes: three wide stacked panels for kinematics, hide repeated upper-row x labels, and include full label bounds in diagnostic PDFs.'}
(D/'manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
print('Layout proof passed for',len(outputs),'figures; automatic limit differences:',axis_changes)
