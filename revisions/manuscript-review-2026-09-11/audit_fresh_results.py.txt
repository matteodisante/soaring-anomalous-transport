"""Check algebraic constraints in outputs of completed stages of the current run.

These checks do not calibrate confidence intervals or establish model adequacy.
They deliberately operate on saved empirical arrays, separately from the generators.
"""
import argparse
import hashlib
import json
from pathlib import Path
import pickle

import numpy as np

ROOT = Path('/Users/matteodisante/local_master_thesis/soaring-anomalous-transport')
RUN = Path('/Volumes/SSD_DISANTE/derived-audit/runs/20260910T221820Z-05d36ab4')
checks = []
inputs = {}

def check(condition, name):
    assert bool(condition), name
    checks.append(name)

def equal(actual, expected, name, rtol=2e-10, atol=1e-9):
    check(np.allclose(actual, expected, rtol=rtol, atol=atol, equal_nan=True), name)

def read(path):
    data = path.read_bytes()
    inputs[str(path)] = hashlib.sha256(data).hexdigest()
    return json.loads(data)

def arr(x):
    return np.asarray(x, dtype=float)

def nonincreasing(x):
    return np.all(np.diff(arr(x)) <= 0)

def slope(x, y):
    x, y = np.log10(arr(x)), np.log10(arr(y))
    xc, yc = x-x.mean(), y-y.mean()
    return float(xc @ yc / (xc @ xc))

def duration():
    data = read(ROOT/'thesis/generated/duration_equipment.json')
    for name in ['paragliders', 'hang gliders']:
        ds = data[name]
        check(nonincreasing([c['flights'] for c in ds['cohorts']]), name+' nested duration counts')
        for c in ds['cohorts']:
            tag = f"{name} T>={c['threshold_s']}"
            n, m = arr(c['support']), arr(c['curve'])
            check(nonincreasing(n) and np.all(n <= c['flights']), tag+' support')
            check(np.all(m[np.isfinite(m)] >= 0), tag+' nonnegative MSD')
    ctrl = data['equipment_control']
    reference = arr(list(ctrl['reference_proportions'].values()))
    equal(reference.sum(), 1, 'reference class proportions sum to one')
    lags = arr(data['paragliders']['lags_s'])
    common = np.isin(lags, ctrl['common_lags_s'])
    check(np.all(lags[common] <= 1800), 'eight-cell common fit respects one-hour half-duration')
    for c in ctrl['cohorts']:
        tag = f"equipment T>={c['threshold_s']}"
        curves = arr([r['curve'] for r in c['classes']])
        support = arr([r['support'] for r in c['classes']])
        check(sum(r['flights'] for r in c['classes'])+c['other_flights']==c['total_flights'], tag+' exhaustive class counts')
        equal(support.sum(axis=0), c['known_class_support'], tag+' summed class support')
        reconstructed = np.divide(
            np.nansum(curves*support, axis=0), support.sum(axis=0),
            out=np.full(len(lags), np.nan), where=support.sum(axis=0)>0)
        equal(reconstructed, arr(c['known_class_raw']), tag+' observed mixture identity')
        fixed = np.sum(reference[:,None]*curves, axis=0)
        fixed[~np.all(support>=30,axis=0)] = np.nan
        equal(fixed, arr(c['standardized']), tag+' fixed mixture identity')
        for row in c['classes']:
            equal(slope(lags[common], arr(row['curve'])[common]), row['exponent'], tag+' '+row['class']+' independent slope')
    c0, c4 = ctrl['cohorts'][0], ctrl['cohorts'][-1]
    equal(arr(c4['known_class_raw'])/arr(c0['known_class_raw']), arr(ctrl['raw_four_hour_ratio']), 'observed four-hour ratio')
    equal(arr(c4['standardized'])/arr(c0['standardized']), arr(ctrl['fixed_mix_four_hour_ratio']), 'standardized four-hour ratio')

def transport():
    data = read(ROOT/'thesis/generated/ch3_revision.json')
    cache_path = RUN/'arrays/ch3_revision_sample.pkl'
    cache_bytes = cache_path.read_bytes()
    inputs[str(cache_path)] = hashlib.sha256(cache_bytes).hexdigest()
    cache = pickle.loads(cache_bytes)  # locally generated by this recorded run
    q = arr(data['q'])
    for name, d in data['results'].items():
        m = cache['measured'][name]
        for count in ['n_per_lag','n_order_two','n_increments']:
            check(nonincreasing(d[count]), name+' nonincreasing '+count)
        check(np.all(arr(d['n_order_two']) <= arr(d['n_per_lag'])), name+' second-difference support')
        km = arr(d['mardia'])
        check(np.all(km[np.isfinite(km)] >= -4-1e-6), name+' covariance-standardized kurtosis Jensen bound')
        c = d['quantile_control']
        quant = arr(c['quantiles'])
        check(np.all(np.diff(quant,axis=-1)[np.isfinite(np.diff(quant,axis=-1))] >= 0), name+' ordered quantiles')
        for axis in [0,1]:
            diff = quant[:,:,2,:]-quant[:,:,axis,:]
            check(np.all(diff[np.isfinite(diff)] >= -1e-8), name+f' radial quantiles dominate component {axis}')
        f, w = arr(c['flights_per_lag']), arr(c['windows_per_lag'])
        equal(f[1:], np.full_like(f[1:], d['n_fixed']), name+' fixed flight counts')
        equal(w[3], np.full_like(w[3], sum(c['fixed_origin_counts'])), name+' fixed origin counts')
        equal(f[1], f[2], name+' same flight support before/after reweighting')
        equal(w[1], w[2], name+' same windows before/after reweighting')
        lags = arr(d['lags_s'])
        common = np.isin(lags, c['fit_lags_s'])
        for method in range(4):
            for axis in range(3):
                for p in range(4):
                    equal(slope(lags[common], quant[method,common,axis,p]), c['exponents'][method][axis][p], name+f' quantile slope {method}/{axis}/{p}')
        moments = arr(m['moments'])
        logm = np.log(moments)
        divided = np.diff(logm,axis=1)/np.diff(q)
        curvature = np.diff(divided,axis=1)
        check(np.all(curvature[np.isfinite(curvature)] >= -1e-7), name+' moment log-convexity at each lag')
        for pca in d['pca']:
            cov = arr(pca['covariance'])
            equal(cov, cov.T, name+' symmetric regional covariance')
            eig = np.linalg.eigvalsh(cov)
            check(eig[0]>0, name+' nonsingular regional covariance')
            equal(eig[1]/eig[0], pca['ratio'], name+' eigenvalue anisotropy ratio')
        for scale, curve in m['coarse'].items():
            equal(curve[0],1,name+f' VACF at zero, h={scale}')
        # A lag-unbiased sample covariance divided by the whole-record variance is
        # not bounded by one at every nonzero lag; no false [-1,1] check is imposed.

def segmentation():
    data = read(ROOT/'thesis/generated/segmentation_decoder_audit.json')
    for name, d in data['disciplines'].items():
        for key in ['fitted_transition_matrix','decoding_transition_matrix']:
            a = arr(d[key])
            check(np.all((a>=0)&(a<=1)), name+' '+key+' probabilities')
            equal(a.sum(axis=1),np.ones(3),name+' '+key+' row sums')
        check(d['archive_decoder_matches_current'],name+' current archive decoder')
        check(d['example']['stored_archive_path_reproduced'],name+' reproduced validation path')
        check(d['selected_restart_diagnostics']['minimum_covariance_eigenvalue']>0,name+' positive fitted covariances')
        c = d['coverage']
        equal(sum(c['seconds_by_reason'].values()),c['cleaned_duration_s'],name+' exhaustive duration coverage')
        equal(sum(c['decisions_by_reason'].values()),c['n_decision_points'],name+' exhaustive decision coverage')
        equal(100*(1-c['seconds_by_reason']['classified']/c['cleaned_duration_s']),c['unclassified_duration_percent'],name+' unclassified time percentage')
        equal(100*(1-c['decisions_by_reason']['classified']/c['n_decision_points']),c['unclassified_decision_percent'],name+' unclassified decision percentage')
        equal(sum(d['example']['counts_current'].values()), d['example']['full_decisions'], name+' example exhaustive phase counts')
