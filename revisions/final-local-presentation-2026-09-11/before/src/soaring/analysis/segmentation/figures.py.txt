"""Diagnostic figures for fitted phase models and their decoded trajectories."""

from __future__ import annotations

from itertools import pairwise
from pathlib import Path

import numpy as np
import pandas as pd

from soaring.reporting.style import PHASE_COLORS

from .features import FEATURE_COLUMNS
from .labels import STATES
from .model import HMMArtifact, effective_transition_matrix


def plot_model_diagnostics(artifact: HMMArtifact, output: str | Path) -> None:
    """Draw transitions, Gaussian emissions, and restart stability."""
    import matplotlib.pyplot as plt

    inverse_mapping = {
        state: component for component, state in artifact.state_mapping.items()
    }
    if set(inverse_mapping) != set(STATES):
        raise ValueError("model diagnostic requires a complete semantic state mapping")
    components = [inverse_mapping[state] for state in STATES]
    transition_matrix = effective_transition_matrix(artifact)[
        np.ix_(components, components)
    ]
    figure, axes = plt.subplots(3, 2, figsize=(6.1, 7.4), constrained_layout=True)
    transition_axis, means_axis, restart_axis = axes[0, 0], axes[0, 1], axes[1, 0]
    covariance_axes = [axes[1, 1], axes[2, 0], axes[2, 1]]
    image = transition_axis.imshow(transition_matrix, vmin=0.0, vmax=1.0, cmap="Blues")
    transition_axis.set(
        xticks=range(len(STATES)),
        yticks=range(len(STATES)),
        xticklabels=["T", "S", "C"],
        yticklabels=["T", "S", "C"],
        xlabel="next phase",
        ylabel="current phase",
        title="Decoding transition matrix",
    )
    for row in range(len(STATES)):
        for col in range(len(STATES)):
            transition_axis.text(
                col,
                row,
                f"{transition_matrix[row, col]:.2f}",
                ha="center",
                va="center",
                color="white" if transition_matrix[row, col] > 0.65 else "black",
            )
    figure.colorbar(image, ax=transition_axis, fraction=0.046)

    x = np.arange(len(FEATURE_COLUMNS))
    width = 0.24
    for offset, (component, name) in enumerate(zip(components, STATES, strict=True)):
        means_axis.bar(
            x + (offset - 1) * width,
            artifact.model.means_[component],
            width=width,
            label=name[0].upper(),
            color=PHASE_COLORS[name],
        )
    means_axis.axhline(0.0, color="black", linewidth=0.7)
    means_axis.set(
        xticks=x,
        xticklabels=[
            r"$\bar v_z$",
            r"$\bar v_h$",
            r"$\overline{|\omega|}$",
            r"$C_\omega$",
        ],
        ylabel="standardized emission mean",
        title="Original 4D emission means",
    )
    means_axis.set_ylim(
        float(np.min(artifact.model.means_)) - 0.15,
        float(np.max(artifact.model.means_)) + 0.55,
    )
    means_axis.legend(
        frameon=False,
        loc="upper center",
        ncol=3,
        fontsize=8.5,
        columnspacing=0.7,
        handlelength=1,
    )

    restart_scores = np.asarray(
        [
            np.nan if score is None else score / max(1, artifact.n_fit_observations)
            for score in artifact.restart_log_likelihoods
        ],
        dtype=float,
    )
    if np.isfinite(restart_scores).any():
        restart_axis.plot(
            np.arange(1, len(restart_scores) + 1),
            restart_scores - np.nanmax(restart_scores),
            marker="o",
            linewidth=1.0,
        )
        restart_axis.axvline(
            artifact.selected_restart + 1,
            color="#C98A1E",
            linestyle="--",
            label="selected",
        )
        restart_axis.legend(frameon=False)
        restart_axis.set(
            xlabel="restart",
            ylabel="Log likelihood relative to best\n(per training decision)",
            title="Initialization stability",
            xticks=np.arange(1, len(restart_scores) + 1),
        )
        restart_axis.ticklabel_format(
            axis="y",
            style="sci",
            scilimits=(-3, 3),
            useOffset=False,
            useMathText=True,
        )
    else:
        restart_axis.set_axis_off()

    short_labels = [r"$\bar v_z$", r"$\bar v_h$", r"$|\omega|$", r"$C_\omega$"]
    covariance_limit = float(np.nanmax(np.abs(artifact.model.covars_)))
    for axis_index, (component, name) in enumerate(
        zip(components, STATES, strict=True)
    ):
        covariance = artifact.model.covars_[component]
        covariance_image = covariance_axes[axis_index].imshow(
            covariance,
            cmap="RdBu_r",
            vmin=-covariance_limit,
            vmax=covariance_limit,
        )
        covariance_axes[axis_index].set(
            xticks=range(len(FEATURE_COLUMNS)),
            yticks=range(len(FEATURE_COLUMNS)),
            xticklabels=short_labels,
            yticklabels=short_labels,
            title=f"{name}: original covariance",
        )
        figure.colorbar(
            covariance_image, ax=covariance_axes[axis_index], fraction=0.046
        )
    naming = (
        "Names calibrated on human training labels"
        if artifact.mapping_method == "manual-train-hungarian"
        else "Provisional state names; human validation pending"
    )
    figure.suptitle(f"Fitted 4D model and decoding transitions\n{naming}", fontsize=11)
    figure.savefig(
        output,
        dpi=180,
        metadata={"CreationDate": None},
        bbox_inches="tight",
        pad_inches=0.08,
    )
    plt.close(figure)


def plot_phase_trajectory(
    points: pd.DataFrame,
    output: str | Path,
    *,
    duration_s: float = 600.0,
    coherence_in_emissions: bool = True,
) -> None:
    """Draw the first ten minutes of one segment on aligned, full-width axes.

    Use the same held-out segment as before, without selecting on phase labels or
    smoothing/decimating the stored observations. Each feature gets its own axis;
    the phase strip and posterior share their clock. This is a display excerpt,
    not a change to decoding, feature windows, or the statistical population.
    """
    import matplotlib.pyplot as plt
    from matplotlib.collections import LineCollection
    from matplotlib.patches import Patch
    from matplotlib.ticker import MaxNLocator, MultipleLocator

    required = {
        "t",
        "z",
        "phase",
        "mean_v_z",
        "mean_v_h",
        "mean_abs_turn_rate",
        "turn_coherence",
        "p_transition",
        "p_search",
        "p_climb",
    }
    missing = required.difference(points.columns)
    if missing:
        raise ValueError(
            f"phase trajectory figure is missing columns: {sorted(missing)}"
        )
    if points.empty or not np.isfinite(duration_s) or duration_s <= 0:
        raise ValueError("phase trajectory needs observations and a positive duration")
    if "segment_id" in points and points["segment_id"].nunique() != 1:
        raise ValueError("phase trajectory must show exactly one preprocessing segment")
    ordered = points.sort_values("t", kind="stable")
    origin = float(ordered["t"].iloc[0])
    ordered = ordered.loc[ordered["t"] <= origin + duration_s]
    elapsed = (ordered["t"].to_numpy(dtype=float) - origin) / 60.0
    palette = {
        **PHASE_COLORS,
        "unclassified": "#9E9E9E",
    }
    colors = [palette[str(phase)] for phase in ordered["phase"]]
    with plt.rc_context(
        {
            "font.size": 10.5,
            "axes.labelsize": 10.5,
            "xtick.labelsize": 10,
            "ytick.labelsize": 10,
            "pdf.fonttype": 42,
        }
    ):
        figure, axes = plt.subplots(
            7,
            1,
            figsize=(6.1, 8.2),
            sharex=True,
            gridspec_kw={"height_ratios": [0.25, 1.0, 0.85, 0.85, 0.85, 0.75, 1.0]},
        )
        figure.subplots_adjust(
            left=0.16, right=0.98, bottom=0.065, top=0.89, hspace=0.20
        )
        identity = (
            str(ordered["flight_id"].iloc[0]) if "flight_id" in ordered else "example"
        )
        figure.suptitle(
            f"Flight {identity} — first {elapsed[-1]:g} min of segment",
            y=0.985,
            fontsize=12,
        )
        figure.legend(
            handles=[
                Patch(color=color, label=phase) for phase, color in palette.items()
            ],
            loc="upper center",
            bbox_to_anchor=(0.53, 0.956),
            ncol=4,
            frameon=False,
            fontsize=9,
            columnspacing=1.1,
            handlelength=1.1,
        )
        dt = np.diff(elapsed)
        cadence = float(np.median(dt)) if dt.size else 1 / 6
        connected = (dt > 0) & (dt <= 1.5 * cadence)
        axes[0].set(ylabel="Phase", yticks=[])
        phases = ordered["phase"].to_numpy()
        boundaries = np.r_[
            0, np.flatnonzero((phases[1:] != phases[:-1]) | ~connected) + 1, len(phases)
        ]
        for first, stop in pairwise(boundaries):
            axes[0].axvspan(
                elapsed[first] - cadence / 2,
                elapsed[stop - 1] + cadence / 2,
                color=colors[first],
                linewidth=0,
            )
            if stop - first >= 3 and phases[first] != "unclassified":
                axes[0].text(
                    (elapsed[first] + elapsed[stop - 1]) / 2,
                    0.5,
                    str(phases[first])[0].upper(),
                    ha="center",
                    va="center",
                    fontsize=8,
                    color="black" if phases[first] == "search" else "white",
                )
        axes[0].spines[["top", "right", "left", "bottom"]].set_visible(False)
        xy = np.column_stack([elapsed, ordered["z"].to_numpy(dtype=float)])
        midpoints = (xy[:-1] + xy[1:]) / 2
        edges = np.concatenate(
            [
                np.stack([xy[:-1], midpoints], axis=1)[connected],
                np.stack([midpoints, xy[1:]], axis=1)[connected],
            ]
        )
        edge_colors = np.concatenate(
            [np.asarray(colors[:-1])[connected], np.asarray(colors[1:])[connected]]
        )
        axes[1].add_collection(
            LineCollection(
                edges,
                colors=edge_colors,
                linewidths=1.5,
            )
        )
        axes[1].autoscale_view()
        axes[1].set_ylabel("Altitude\n(m)")
        features = [
            ("mean_v_z", "Vertical speed\n(m/s)", "#4E8A5B", 1.0),
            ("mean_v_h", "Horizontal speed\n(m/s)", "#3477A8", 1.0),
            (
                "mean_abs_turn_rate",
                "Mean absolute\nturn rate (deg/s)",
                "#B66A00",
                180 / np.pi,
            ),
            (
                "turn_coherence",
                "Turn\ncoherence"
                if coherence_in_emissions
                else "Coherence\n(diagnostic)",
                "#555555",
                1.0,
            ),
        ]
        # NaNs break feature/posterior curves across unavailable observations.
        for axis, (column, label, color, factor) in zip(
            axes[2:6], features, strict=True
        ):
            values = ordered[column].to_numpy(dtype=float).copy() * factor
            values[1:][~connected] = np.nan
            axis.plot(elapsed, values, color=color, linewidth=1.25)
            axis.set_ylabel(label)
        axes[2].axhline(0, color="0.5", linewidth=0.7, linestyle="--")
        axes[5].set(ylim=(-0.05, 1.05), yticks=[0, 0.5, 1])
        for state, line_style in zip(STATES, ("-", "--", "-."), strict=True):
            values = ordered[f"p_{state}"].to_numpy(dtype=float).copy()
            values[1:][~connected] = np.nan
            axes[6].plot(
                elapsed,
                values,
                color=palette[state],
                linewidth=1.35,
                linestyle=line_style,
            )
        axes[6].set(
            ylabel="Posterior\nprobability",
            ylim=(-0.05, 1.05),
            yticks=[0, 0.5, 1],
            xlabel="Time from segment start (min)",
        )
        for i, axis in enumerate(axes[1:]):
            axis.text(
                0.012,
                0.83,
                f"({chr(97 + i)})",
                transform=axis.transAxes,
                fontsize=9,
                bbox={
                    "facecolor": "white",
                    "edgecolor": "none",
                    "alpha": 0.8,
                    "pad": 1,
                },
            )
            axis.spines[["top", "right"]].set_visible(False)
            axis.grid(axis="x", color=".9", linewidth=0.6)
            if i < 4:
                axis.yaxis.set_major_locator(MaxNLocator(nbins=3))
        axes[-1].xaxis.set_major_locator(MultipleLocator(1))
        axes[-1].set_xlim(0, max(float(elapsed[-1]), 1 / 6))
        figure.savefig(output, dpi=180, metadata={"CreationDate": None})
        plt.close(figure)


def plot_confusion_matrix(
    truth: np.ndarray, prediction: np.ndarray, output: str | Path, *, title: str
) -> None:
    """Draw an integer confusion matrix in the fixed semantic phase order."""
    import matplotlib.pyplot as plt

    matrix = np.zeros((len(STATES), len(STATES)), dtype=int)
    for row, expected in enumerate(STATES):
        for col, inferred in enumerate(STATES):
            matrix[row, col] = int(
                np.sum((truth == expected) & (prediction == inferred))
            )
    figure, axis = plt.subplots(figsize=(4.5, 4.0), constrained_layout=True)
    image = axis.imshow(matrix, cmap="Blues")
    axis.set(
        xticks=range(len(STATES)),
        yticks=range(len(STATES)),
        xticklabels=STATES,
        yticklabels=STATES,
        xlabel="inferred phase",
        ylabel="manual phase",
        title=title,
    )
    for row in range(len(STATES)):
        for col in range(len(STATES)):
            axis.text(col, row, str(matrix[row, col]), ha="center")
    figure.colorbar(image, ax=axis, fraction=0.046)
    figure.savefig(output, dpi=180)
    plt.close(figure)
