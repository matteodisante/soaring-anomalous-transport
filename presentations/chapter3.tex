\begin{frame}{Constant velocity cancels; an intrinsic $H$ remains open}
\[
 A_2\mathbf r=\mathbf r(t+2\tau)-2\mathbf r(t+\tau)+\mathbf r(t),
 \qquad V_2=\langle|A_2\mathbf r|^2\rangle.
\]
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\textbf{Exact cancellation}\par
\[
 \mathbf r(t)=\mathbf r_0+\mathbf v_dt+\mathbf Y(t)
 \quad\Rightarrow\quad A_2\mathbf r=A_2\mathbf Y.
\]
\small The constant velocity may combine wind, route choice and net progress. A turn or spatially varying wind survives.
\column{.48\textwidth}
\textbf{Conditional interpretation}\par
\small If centred increments have finite variance, stationary statistics and the law $K\tau^{2H}$,
\[
 V_2=K(4-2^{2H})\tau^{2H},\quad 0<H<1.
\]
At $H=1$ the prefactor vanishes. Our $b_2\simeq2$ cannot be read as a validated $H\simeq1$.
\end{columns}
\question{Which residual and which scale interval could satisfy the assumptions needed to estimate $H$?}
\source{At $\tau=10^4$ s, $V_2$ retains only \num{\StatRevParaLastOrderTwoFlights} para / \num{\StatRevHangLastOrderTwoFlights} hang flights; it needs $2\tau$ of continuous support.}
\speaker{Invariant means unchanged after adding a constant position offset and a constant velocity times time. It does not mean constant as tau changes. For adjacent increments I1 and I2, their sum spans 2tau and their difference is A2. The parallelogram identity gives V2=4M(tau)-M(2tau) for the same stationary-increment population. Substituting M(tau)=K tau to the power 2H gives the displayed formula. All three conditions matter: finite variance, stationary increment statistics, and the same power law through 2tau. Stationarity means that the increment law at a given duration does not change with the start time; it does not require independence. A fitted b2 above two does not by itself identify which assumption fails. A random-phase circular trajectory has stationary increments and finite variance but no exact increment power law, and V2 grows as tau to the fourth at small lag.}
\end{frame}

\begin{frame}{Second differences measure changes of interval-mean velocity}
\[
 \overline{\mathbf v}_1=\frac{\mathbf r(t+\tau)-\mathbf r(t)}{\tau},\qquad
 \overline{\mathbf v}_2=\frac{\mathbf r(t+2\tau)-\mathbf r(t+\tau)}{\tau}.
\]
\[
 \frac{A_2\mathbf r}{\tau}=\overline{\mathbf v}_2-\overline{\mathbf v}_1,
 \qquad D_2(\tau)=\frac{\sqrt{V_2(\tau)}}{\tau}
 =\sqrt{\left\langle|\overline{\mathbf v}_2-\overline{\mathbf v}_1|^2\right\rangle}.
\]
\small $D_2$ is in m/s. At $\tau=60$ s it compares the mean velocity vector in
one minute with the vector in the next minute. Changes of magnitude and direction both contribute.
\question{At which scales does the ground motion change most, and how does that depend on the region?}
\source{Exact kinematic identity; no power law or stationarity assumption is needed.}
\speaker{This is an RMS change of interval-mean ground velocity. It is neither instantaneous acceleration nor wind speed. A constant added velocity cancels in both intervals. Keeping the physical units makes regional amplitudes interpretable even when a single exponent is not supported. The later regional study uses this quantity together with the vector covariance, distributions and order-three control.}
\end{frame}

\begin{frame}{Third differences also cancel a constant acceleration}
\[
 A_3\mathbf r=\mathbf r(t+3\tau)-3\mathbf r(t+2\tau)
               +3\mathbf r(t+\tau)-\mathbf r(t),\qquad
 V_3=\langle|A_3\mathbf r|^2\rangle.
\]
\small For $\mathbf r=\mathbf r_0+\mathbf v_0t+\tfrac12\mathbf a t^2$ with constant $\mathbf a$,
\[
 V_2=|\mathbf a|^2\tau^4,\qquad V_3=0.
\]
Under the centred stationary-increment power law,
\[
 \frac{V_3}{V_2}=\frac{15-6\,2^{2H}+3^{2H}}{4-2^{2H}},\qquad 0<H<1.
\]
\question{Do the same supported observations satisfy the predicted relation between orders?}
\source{Order three requires $3\tau$ of continuous support; the model law must hold through $3\tau$.}
\speaker{V3 is a sensitivity check to quadratic position trends. A smooth turn survives: typically V2 scales as tau to the fourth and V3 as tau to the sixth at small lag. Under the stated model both orders have exponent 2H but different amplitudes; Brownian motion plus constant velocity gives V3/V2=3. Independent position error with variance sigma squared per coordinate contributes 6 sigma squared at order two and 20 at order three. Correlated errors and smoothing require a different noise calculation. Agreement between orders does not prove that environmental forcing has been removed.}
\end{frame}

\section{Quantiles and population control}

\begin{frame}{A single displacement scale predicts equal quantile slopes}
\begin{columns}[T,onlytextwidth]
\column{.53\textwidth}
\[
 Q_p(\tau)=\inf\{r:F_\tau(r)\ge p\}.
\]
\small An $H$-self-similar process with stationary increments predicts
\[
 \begin{aligned}
 R(\tau)&\overset d=(\tau/\tau_0)^H R(\tau_0),\\
 Q_p(\tau)&=(\tau/\tau_0)^H Q_p(\tau_0).
 \end{aligned}
\]
\[
 H_p=H\ \text{for every }p;\quad Q_{.90}/Q_{.25}\text{ stays fixed}.
\]

\section{Vector scaling and dependence}

\begin{frame}{Anisotropy does not prevent a common scalar rescaling}
\[
 \mathbf X(\tau)\overset d=\tau^H\mathbf Z
 \quad\Longrightarrow\quad
 |\mathbf X(\tau)|\overset d=\tau^H|\mathbf Z|,
\]
\small even if the law of $\mathbf Z$ is anisotropic.
\par\medskip
With two centred independent Gaussian components instead,
\[
 \mathbb E[R^2]=a\tau^{2H_E}+b\tau^{2H_N},
\]
unequal component exponents can produce a lag-dependent radial effective exponent.
\question{Is a mismatch caused by unequal directional scales, changing marginal shape or changing joint structure?}
\speaker{Scalar self-similarity and isotropy are separate hypotheses. Operator self-similarity allows a matrix exponent and is broader, but does not permit arbitrary lag-by-lag normalization chosen to force collapse. The current plots assess the stated scalar comparison; they do not exhaust that wider model class.}
\end{frame}

\begin{frame}{East, north and radius together still do not identify the joint law}
\[
 \{(1,2),(-1,-2)\}
 \qquad\text{versus}\qquad
 \{(1,-2),(-1,2)\},
\]
\small each pair equiprobable.

Signed north marginal & $\pm2$ & $\pm2$\\
Radius & $\sqrt5$ & $\sqrt5$\\

\begin{frame}{The moment spectrum challenges a specific L\'evy-walk benchmark}

\[

\begin{frame}{Non-Gaussian pooled increments can arise from Gaussian mixtures}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}\fig[48mm]{talk-kurtosis.pdf}

\[
 \begin{aligned}
 \mathbf Z&=\Sigma^{-1/2}(\mathbf X-\bm\mu),\\

\]

\begin{frame}[label=memory-backup]{Finite velocity persistence gives a ballistic-to-diffusive crossover}
\small
For the same stationary, centred velocity population,
\[
 M_{2,c}(\tau)=2\sigma_v^2\int_0^\tau(\tau-s)C(s)\,ds.
\]
With $C(s)=e^{-s/t_p}$,
\[
 M_{2,c}=2\sigma_v^2t_p[\tau-t_p(1-e^{-\tau/t_p})].
\]
Ballistic at short times, diffusive at long times.

\source{\href{https://doi.org/10.1112/plms/s2-20.1.196}{Taylor (1922)}, pp. 207--208. This covariance identity cannot be applied directly to differently weighted heterogeneous means.}
\end{frame}
